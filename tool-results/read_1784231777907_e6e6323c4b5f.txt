    16→    16→## 1. Architecture Overview
    17→    17→
    18→    18→### The Three-Pillar Model
    19→    19→
    20→    20→```
    21→    21→+-----------------------------------------------------------+
    22→    22→|                    CATALOG BUILDER                         |
    23→    23→|                                                            |
    24→    24→|  +-------------+  +-------------+  +--------------------+  |
    25→    25→|  |   PILLAR 1  |  |   PILLAR 2  |  |     PILLAR 3       |  |
    26→    26→|  |    DATA     |  |   LAYOUT    |  |     SETTINGS       |  |
    27→    27→|  |             |  |             |  |                     |  |
    28→    28→|  | DataSources |  | Collections |  | Language (FR/EN/AR) |  |
    29→    29→|  | Columns     |  | Components  |  | Admin Access        |  |
    30→    30→|  | Rows        |  | Mappings    |  | Color Theme         |  |
    31→    31→|  | Relations   |  | Preview     |  | Publication         |  |
    32→    32→|  | Arrays      |  |             |  | Sharing             |  |
    33→    33→|  +-------------+  +-------------+  +--------------------+  |
    34→    34→|                                                            |
    35→    35→+----------------------------|-------------------------------+
    36→    36→                             |
    37→    37→                    +--------v--------+
    38→    38→                    |  PUBLIC CATALOG  |
    39→    39→                    |  (Read-Only)     |
    40→    40→                    |  Renders from    |
    41→    41→                    |  Layout + Data   |
    42→    42→                    +------------------+
    43→    43→```
    44→    44→
    45→    45→### Core Design Principle: Separation of Schema from Storage
    46→    46→
    47→    47→The fundamental insight is that **user-defined columns are metadata**, and **row data is a JSON blob keyed by column IDs**. This avoids the need for ALTER TABLE on every column change and makes SQLite viable.
    48→    48→
    49→    49→```
    50→    50→Traditional:  ALTER TABLE products ADD COLUMN color TEXT;
    51→    51→Our approach:  columns table (metadata) + rows table (JSON data blob)
    52→    52→```

    56→    56→## 2. Prisma Schema — Complete
    57→    57→
    58→    58→```prisma

    69→    69→// PILLAR 1: DATA — Dynamic Data Sources, Columns, Rows
    70→    70→// ═══════════════════════════════════════════════════════════════
    71→    71→
    72→    72→/// A user-defined table (e.g., "Products", "Colors", "Categories")
    73→    73→model DataSource {
    74→    74→  id          String    @id @default(cuid())
    75→    75→  name        String    @unique          // "Products", "Colors", etc.
    76→    76→  slug        String    @unique          // URL-safe identifier
    77→    77→  icon        String    @default("table") // Lucide icon name
    78→    78→  color       String    @default("#6366f1") // Accent color for UI
    79→    79→  sortOrder   Int       @default(0)      // Display order in sidebar
    80→    80→  createdAt   DateTime  @default(now())
    81→    81→  updatedAt   DateTime  @updatedAt
    82→    82→
    83→    83→  // Relations
    84→    84→  columns     Column[]
    85→    85→  rows        Row[]
    86→    86→  sourceOf    Relation[] @relation("SourceTable")   // This table is the source in a relation
    87→    87→  targetOf    Relation[] @relation("TargetTable")   // This table is the target in a relation
    88→    88→
    89→    89→  @@map("data_sources")
    90→    90→}
    91→    91→
    92→    92→/// A column definition within a DataSource — pure metadata
    93→    93→model Column {
    94→    94→  id           String     @id @default(cuid())
    95→    95→  dataSourceId String
    96→    96→  name         String                       // "Price", "Image URL", etc.
    97→    97→  slug         String                       // URL-safe: "price", "image_url"
    98→    98→  type         ColumnType  @default(TEXT)
    99→    99→  sortOrder    Int        @default(0)       // Display order in table
   100→   100→  required     Boolean    @default(false)   // Is this column required?
   101→   101→  visible      Boolean    @default(true)    // Show in table view?
   102→   102→  defaultValue String?                      // JSON-encoded default value
   103→   103→
   104→   104→  // For ARRAY type: which child columns compose this array
   105→   105→  arrayChildIds String?                     // JSON array of column IDs: ["col_1","col_2","col_3"]
   106→   106→
   107→   107→  // For RELATION type: which table/column this links to
   108→   108→  relationId   String?                      // FK to Relation record (if type = RELATION)
   109→   109→
   110→   110→  createdAt    DateTime   @default(now())
   111→   111→  updatedAt    DateTime   @updatedAt
   112→   112→
   113→   113→  dataSource   DataSource @relation(fields: [dataSourceId], references: [id], onDelete: Cascade)
   114→   114→
   115→   115→  @@unique([dataSourceId, slug])
   116→   116→  @@map("columns")
   117→   117→}
   118→   118→
   119→   119→/// Column types — determines how data is stored, validated, and rendered
   120→   120→enum ColumnType {
   121→   121→  TEXT          // Short text (single line)
   122→   122→  LONG_TEXT     // Multi-line text / description
   123→   123→  NUMBER        // Numeric (int or float)
   124→   124→  CURRENCY      // Number formatted as price
   125→   125→  IMAGE         // Single image URL
   126→   126→  IMAGE_ARRAY   // JSON array of image URLs (standalone, not composed)
   127→   127→  COLOR         // Color swatch: JSON { name, hex }
   128→   128→  BOOLEAN       // true/false
   129→   129→  DATE          // ISO date string
   130→   130→  URL           // Web URL
   131→   131→  PHONE         // Phone number
   132→   132→  EMAIL         // Email address
   133→   133→  SELECT        // Single select from predefined options
   134→   134→  MULTI_SELECT  // Multi select from predefined options
   135→   135→  ARRAY         // Virtual column composed of other columns (arrayChildIds)
   136→   136→  RELATION      // Links to a row in another DataSource
   137→   137→}
   138→   138→
   139→   139→/// A row of data — stores all cell values in a single JSON object keyed by column ID
   140→   140→model Row {
   141→   141→  id           String     @id @default(cuid())
   142→   142→  dataSourceId String
   143→   143→  data         String     // JSON: { "col_abc123": "value1", "col_def456": 42, ... }
   144→   144→  sortOrder    Int        @default(0)
   145→   145→  createdAt    DateTime   @default(now())
   146→   146→  updatedAt    DateTime   @updatedAt
   147→   147→
   148→   148→  dataSource   DataSource @relation(fields: [dataSourceId], references: [id], onDelete: Cascade)
   149→   149→
   150→   150→  @@map("rows")
   151→   151→}
   152→   152→
   153→   153→/// A relation between two DataSources (like a foreign key)
   154→   154→model Relation {
   155→   155→  id             String     @id @default(cuid())
   156→   156→  name           String                       // "Product → Category"
   157→   157→  sourceColumnId String                       // The column in source table that holds the reference
   158→   158→  sourceTableId  String                       // The source DataSource
   159→   159→  targetTableId  String                       // The target DataSource
   160→   160→  type           RelationType @default(MANY_TO_ONE)
   161→   161→
   162→   162→  sourceColumn   Column     @relation(fields: [sourceColumnId], references: [id], onDelete: Cascade)
   163→   163→  sourceTable    DataSource @relation("SourceTable", fields: [sourceTableId], references: [id], onDelete: Cascade)
   164→   164→  targetTable    DataSource @relation("TargetTable", fields: [targetTableId], references: [id], onDelete: Cascade)
   165→   165→
   166→   166→  @@map("relations")
   167→   167→}
   168→   168→
   169→   169→enum RelationType {
   170→   170→  MANY_TO_ONE    // Many rows in source → one row in target (e.g., products → category)
   171→   171→  ONE_TO_MANY    // One row in source → many rows in target (e.g., category → products)
   172→   172→  MANY_TO_MANY   // Junction (future: via junction table)
   173→   173→}
   174→   174→
   175→   175→/// Select options for SELECT/MULTI_SELECT column types
   176→   176→model SelectOption {
   177→   177→  id         String   @id @default(cuid())
   178→   178→  columnId   String
   179→   179→  label      String
   180→   180→  value      String   // Stored value (can differ from label)
   181→   181→  color      String?  // Hex color for the tag
   182→   182→  sortOrder  Int      @default(0)
   183→   183→
   184→   184→  column     Column   @relation(fields: [columnId], references: [id], onDelete: Cascade)
   185→   185→
   186→   186→  @@unique([columnId, value])
   187→   187→  @@map("select_options")
   188→   188→}

   388→   388→### Schema Design Rationale
   389→   389→
   390→   390→| Decision | Why |
   391→   391→|----------|-----|
   392→   392→| **Row.data is JSON string** | SQLite doesn't support JSON columns natively in Prisma, but `String` works fine with `JSON.parse/stringify`. Columns are metadata; rows are a flat JSON blob keyed by `columnId`. |
   393→   393→| **Column.arrayChildIds is JSON** | An ARRAY column groups multiple real columns. The `arrayChildIds` field stores `["col_a","col_b","col_c"]` so we know which sub-columns to compose. |
   394→   394→| **Relation is a first-class entity** | Instead of burying FK info inside column config, Relations are their own table. This makes validation, resolution, and future many-to-many easier. |
   395→   395→| **Component.mappings is JSON** | Each component type has different mapping needs. `CARD_GRID` maps `{ title, description, coverImage }`, while `IMAGE_CAROUSEL` maps `{ images }`. Storing as flexible JSON avoids a table per component type. |
   396→   396→| **Catalog → Page → Section → Component** | Four-level nesting mirrors how visual editors work. This is the same pattern Glide uses: App → Screen → Container → Element. |
   397→   397→| **CatalogSettings is 1:1** | Separate model avoids bloating the Catalog model. One-to-one via `@unique` on `catalogId`. |
   398→   398→| **Theme is separate** | Theme values change independently and may be swapped/shared in future. |

   402→   402→## 3. API Route Structure
   403→   403→
   404→   404→```
   405→   405→src/app/api/
   406→   406→├── auth/
   407→   407→│   ├── route.ts              # POST login, GET session check, DELETE logout
   408→   408→│
   409→   409→├── data-sources/
   410→   410→│   ├── route.ts              # GET (list), POST (create)
   411→   411→│   ├── [id]/
   412→   412→│   │   ├── route.ts          # GET, PATCH, DELETE a DataSource
   413→   413→│   │   ├── columns/
   414→   414→│   │   │   ├── route.ts      # GET (list columns), POST (add column)
   415→   415→│   │   │   ├── [columnId]/
   416→   416→│   │   │   │   └── route.ts  # PATCH, DELETE a column
   417→   417→│   │   ├── rows/
   418→   418→│   │   │   ├── route.ts      # GET (list rows, paginated), POST (add row)
   419→   419→│   │   │   ├── [rowId]/
   420→   420→│   │   │   │   └── route.ts  # GET, PATCH, DELETE a row
   421→   421→│   │   ├── import/
   422→   422→│   │   │   └── route.ts      # POST CSV import
   423→   423→│   │   ├── export/
   424→   424→│   │   │   └── route.ts      # GET CSV export
   425→   425→│   │   └── relations/
   426→   426→│   │       ├── route.ts      # GET, POST relations for this source
   427→   427→│   │       ├── [relationId]/
   428→   428→│   │       │   └── route.ts  # PATCH, DELETE a relation
   429→   429→│
   430→   430→├── catalogs/
   431→   431→│   ├── route.ts              # GET (list), POST (create)
   432→   432→│   ├── [catalogId]/
   433→   433→│   │   ├── route.ts          # GET, PATCH, DELETE
   434→   434→│   │   ├── pages/
   435→   435→│   │   │   ├── route.ts      # GET, POST
   436→   436→│   │   │   ├── [pageId]/
   437→   437→│   │   │   │   ├── route.ts  # GET, PATCH, DELETE
   438→   438→│   │   │   │   ├── sections/
   439→   439→│   │   │   │   │   ├── route.ts       # GET, POST
   440→   440→│   │   │   │   │   ├── [sectionId]/
   441→   441→│   │   │   │   │   │   ├── route.ts   # GET, PATCH, DELETE
   442→   442→│   │   │   │   │   │   ├── components/
   443→   443→│   │   │   │   │   │   │   ├── route.ts       # GET, POST

   838→   838→## 6. Data Flow Diagrams
   839→   839→
   840→   840→### 6.1 CSV Import Flow
   841→   841→
   842→   842→```
   843→   843→User drops CSV file
   844→   844→        │
   845→   845→        ▼
   846→   846→[Frontend: ImportCSVModal]
   847→   847→        │  POST /api/data-sources/[id]/import
   848→   848→        │  FormData { file }
   849→   849→        ▼
   850→   850→[API: import/route.ts]
   851→   851→        │
   852→   852→        ├── 1. Read CSV with papaparse
   853→   853→        │      Parse headers → detect types
   854→   854→        │        - If header contains "image"/"photo"/"img" → IMAGE
   855→   855→        │        - If header contains "price"/"prix"/"cost"  → CURRENCY
   856→   856→        │        - If all values are numeric                  → NUMBER
   857→   857→        │        - Otherwise                                  → TEXT
   858→   858→        │
   859→   859→        ├── 2. Create DataSource (if new) or use existing
   860→   860→        │
   861→   861→        ├── 3. Create Column records
   862→   862→        │      for each CSV header:
   863→   863→        │        db.column.create({ name, slug, type, dataSourceId })
   864→   864→        │
   865→   865→        ├── 4. Create Row records
   866→   866→        │      for each CSV row:
   867→   867→        │        const data = {};
   868→   868→        │        for each header/column:
   869→   869→        │          data[column.id] = csvRow[header];
   870→   870→        │        db.row.create({ dataSourceId, data: JSON.stringify(data) })
   871→   871→        │
   872→   872→        └── 5. Return { imported: N, columns: [...], skipped: M }
   873→   873→                │
   874→   874→                ▼
   875→   875→[Frontend: Update data-store]
   876→   876→        │  setColumns(newColumns)
   877→   877→        │  setRows(newRows)
   878→   878→        ▼
   879→   879→[TableGrid re-renders]
   880→   880→```
   881→   881→
   882→   882→### 6.2 Layout Configuration Flow
   883→   883→
   884→   884→```
   885→   885→User clicks "Layout" pillar
   886→   886→        │
   887→   887→        ▼
   888→   888→[BuilderMain shows PageManager]
   889→   889→        │
   890→   890→        User clicks a Page → setActivePageId(pageId)
   891→   891→        │
   892→   892→        ▼
   893→   893→[PageEditor loads]
   894→   894→        │  GET /api/catalogs/[id]/pages/[pageId]
   895→   895→        │  Returns: { sections: [...], components: [...] }
   896→   896→        ▼

  1189→  1189→### 7.5 Column-to-Component Mapping Resolution
  1190→  1190→
  1191→  1191→This is the core rendering engine. Given a component's `mappings` JSON and a row's `data` JSON, resolve what to display:
  1192→  1192→
  1193→  1193→```typescript
  1194→  1194→// src/lib/resolve-mappings.ts
  1195→  1195→interface MappingConfig {
  1196→  1196→  dataSourceId: string;
  1197→  1197→  title?: string;        // column ID
  1198→  1198→  subtitle?: string;     // column ID
  1199→  1199→  description?: string;  // column ID
  1200→  1200→  coverImage?: string;   // column ID
  1201→  1201→  price?: string;        // column ID
  1202→  1202→  carouselImages?: string;  // column ID (must be ARRAY or IMAGE_ARRAY type)
  1203→  1203→  colorVariants?: string;   // column ID
  1204→  1204→  sizeVariants?: string;    // column ID
  1205→  1205→  filterBy?: string;        // column ID (for category filter)
  1206→  1206→  sortBy?: string;          // column ID
  1207→  1207→}
  1208→  1208→
  1209→  1209→function resolveCardData(mapping: MappingConfig, row: Row, columns: Column[]) {
  1210→  1210→  const data = JSON.parse(row.data);
  1211→  1211→  const colMap = Object.fromEntries(columns.map(c => [c.id, c]));
  1212→  1212→
  1213→  1213→  return {
  1214→  1214→    id: row.id,
  1215→  1215→    title: data[mapping.title ?? ''] ?? '',
  1216→  1216→    subtitle: resolveValue(data, mapping.subtitle, colMap),
  1217→  1217→    description: resolveValue(data, mapping.description, colMap),
  1218→  1218→    coverImage: resolveValue(data, mapping.coverImage, colMap),
  1219→  1219→    price: mapping.price ? {
  1220→  1220→      value: Number(data[mapping.price] ?? 0),
  1221→  1221→      currency: colMap[mapping.price]?.type === 'CURRENCY',
  1222→  1222→    } : null,
  1223→  1223→  };
  1224→  1224→}
  1225→  1225→
  1226→  1226→function resolveDetailData(mapping: MappingConfig, row: Row, columns: Column[]) {
  1227→  1227→  const data = JSON.parse(row.data);
  1228→  1228→  const colMap = Object.fromEntries(columns.map(c => [c.id, c]));
  1229→  1229→  const card = resolveCardData(mapping, row, columns);
  1230→  1230→
  1231→  1231→  // Resolve array columns
  1232→  1232→  const carouselImages = mapping.carouselImages
  1233→  1233→    ? resolveArrayColumn(data, mapping.carouselImages, colMap)
  1234→  1234→    : [];
  1235→  1235→
  1236→  1236→  const colorVariants = mapping.colorVariants
  1237→  1237→    ? JSON.parse(data[mapping.colorVariants] || '[]')
  1238→  1238→    : [];
  1239→  1239→
  1240→  1240→  const sizeVariants = mapping.sizeVariants
  1241→  1241→    ? JSON.parse(data[mapping.sizeVariants] || '[]')
  1242→  1242→    : [];
  1243→  1243→
  1244→  1244→  return { ...card, carouselImages, colorVariants, sizeVariants };
  1245→  1245→}
  1246→  1246→
  1247→  1247→function resolveArrayColumn(
  1248→  1248→  data: Record<string, unknown>,
  1249→  1249→  arrayColumnId: string,
  1250→  1250→  colMap: Record<string, Column>
  1251→  1251→): string[] {
  1252→  1252→  const col = colMap[arrayColumnId];
  1253→  1253→  if (!col) return [];
  1254→  1254→
  1255→  1255→  if (col.type === 'IMAGE_ARRAY') {
  1256→  1256→    return JSON.parse((data[arrayColumnId] as string) || '[]');
  1257→  1257→  }
  1258→  1258→
  1259→  1259→  if (col.type === 'ARRAY') {
  1260→  1260→    const childIds: string[] = JSON.parse(col.arrayChildIds || '[]');
  1261→  1261→    return childIds.map(id => data[id]).filter(Boolean) as string[];
  1262→  1262→  }
  1263→  1263→
  1264→  1264→  return [];
  1265→  1265→}
  1266→  1266→```

  1303→  1303→### 7.7 Debounced Batch Save for Inline Editing
  1304→  1304→
  1305→  1305→```typescript
  1306→  1306→// src/lib/stores/data-store.ts (excerpt)
  1307→  1307→let flushTimeout: NodeJS.Timeout | null = null;
  1308→  1308→
  1309→  1309→markDirty: (rowId, colId, value) => {
  1310→  1310→  set(state => {
  1311→  1311→    const newDirty = new Map(state.dirtyCells);
  1312→  1312→    newDirty.set(`${rowId}:${colId}`, value);
  1313→  1313→
  1314→  1314→    // Optimistic local update
  1315→  1315→    const newRows = state.rows.map(r => {
  1316→  1316→      if (r.id !== rowId) return r;
  1317→  1317→      const data = JSON.parse(r.data);
  1318→  1318→      data[colId] = value;
  1319→  1319→      return { ...r, data: JSON.stringify(data) };
  1320→  1320→    });
  1321→  1321→
  1322→  1322→    return { dirtyCells: newDirty, rows: newRows };
  1323→  1323→  });
  1324→  1324→
  1325→  1325→  // Debounce flush
  1326→  1326→  if (flushTimeout) clearTimeout(flushTimeout);
  1327→  1327→  flushTimeout = setTimeout(() => get().flushDirty(), 500);
  1328→  1328→},
  1329→  1329→
  1330→  1330→flushDirty: async () => {
  1331→  1331→  const { dirtyCells, rows } = get();
  1332→  1332→  if (dirtyCells.size === 0) return;
  1333→  1333→
  1334→  1334→  // Group by rowId
  1335→  1335→  const byRow = new Map<string, Record<string, string>>();
  1336→  1336→  for (const [key, value] of dirtyCells) {
  1337→  1337→    const [rowId, colId] = key.split(':');
  1338→  1338→    if (!byRow.has(rowId)) byRow.set(rowId, {});
  1339→  1339→    byRow.get(rowId)![colId] = value;
  1340→  1340→  }
  1341→  1341→
  1342→  1342→  // Batch PATCH each row
  1343→  1343→  await Promise.all(
  1344→  1344→    Array.from(byRow.entries()).map(([rowId, updates]) =>
  1345→  1345→      fetch(`/api/data-sources/${get().activeDataSourceId}/rows/${rowId}`, {
  1346→  1346→        method: 'PATCH',
  1347→  1347→        headers: { 'Content-Type': 'application/json' },
  1348→  1348→        body: JSON.stringify({ data: updates }),
  1349→  1349→      })
  1350→  1350→    )
  1351→  1351→  );
  1352→  1352→
  1353→  1353→  set({ dirtyCells: new Map() });
  1354→  1354→},
  1355→  1355→```

  1357→  1357→### 7.8 Relation Resolution on the API Side
  1358→  1358→
  1359→  1359→```typescript
  1360→  1360→// src/app/api/public/[catalogSlug]/product/[rowId]/route.ts
  1361→  1361→export async function GET(req: NextRequest, { params }: { params: { rowId: string } }) {
  1362→  1362→  const row = await db.row.findUnique({ where: { id: params.rowId } });
  1363→  1363→  if (!row) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  1364→  1364→
  1365→  1365→  const data = JSON.parse(row.data);
  1366→  1366→
  1367→  1367→  // Resolve all RELATION columns
  1368→  1368→  const columns = await db.column.findMany({
  1369→  1369→    where: { dataSourceId: row.dataSourceId, type: 'RELATION' },
  1370→  1370→    include: { relation: true },
  1371→  1371→  });
  1372→  1372→
  1373→  1373→  const resolvedRelations: Record<string, any> = {};
  1374→  1374→
  1375→  1375→  for (const col of columns) {
  1376→  1376→    if (!col.relation) continue;
  1377→  1377→    const targetRowId = data[col.id];
  1378→  1378→    if (!targetRowId) continue;
  1379→  1379→
  1380→  1380→    const targetRow = await db.row.findUnique({ where: { id: targetRowId } });
  1381→  1381→    if (!targetRow) continue;
  1382→  1382→
  1383→  1383→    const targetColumns = await db.column.findMany({
  1384→  1384→      where: { dataSourceId: col.relation.targetTableId },
  1385→  1385→    });
  1386→  1386→
  1387→  1387→    const targetData = JSON.parse(targetRow.data);
  1388→  1388→    const resolved: Record<string, any> = {};
  1389→  1389→    for (const tc of targetColumns) {
  1390→  1390→      resolved[tc.slug] = targetData[tc.id];
  1391→  1391→    }
  1392→  1392→    resolvedRelations[col.id] = resolved;
  1393→  1393→  }
  1394→  1394→
  1395→  1395→  return NextResponse.json({
  1396→  1396→    ...data,
  1397→  1397→    _id: row.id,
  1398→  1398→    _relations: resolvedRelations,
  1399→  1399→  });
  1400→  1400→}
