    16→## 1. Architecture Overview
    17→
    18→### The Three-Pillar Model
    19→
    20→```
    21→+-----------------------------------------------------------+
    22→|                    CATALOG BUILDER                         |
    23→|                                                            |
    24→|  +-------------+  +-------------+  +--------------------+  |
    25→|  |   PILLAR 1  |  |   PILLAR 2  |  |     PILLAR 3       |  |
    26→|  |    DATA     |  |   LAYOUT    |  |     SETTINGS       |  |
    27→|  |             |  |             |  |                     |  |
    28→|  | DataSources |  | Collections |  | Language (FR/EN/AR) |  |
    29→|  | Columns     |  | Components  |  | Admin Access        |  |
    30→|  | Rows        |  | Mappings    |  | Color Theme         |  |
    31→|  | Relations   |  | Preview     |  | Publication         |  |
    32→|  | Arrays      |  |             |  | Sharing             |  |
    33→|  +-------------+  +-------------+  +--------------------+  |
    34→|                                                            |
    35→+----------------------------|-------------------------------+
    36→                             |
    37→                    +--------v--------+
    38→                    |  PUBLIC CATALOG  |
    39→                    |  (Read-Only)     |
    40→                    |  Renders from    |
    41→                    |  Layout + Data   |
    42→                    +------------------+
    43→```
    44→
    45→### Core Design Principle: Separation of Schema from Storage
    46→
    47→The fundamental insight is that **user-defined columns are metadata**, and **row data is a JSON blob keyed by column IDs**. This avoids the need for ALTER TABLE on every column change and makes SQLite viable.
    48→
    49→```
    50→Traditional:  ALTER TABLE products ADD COLUMN color TEXT;
    51→Our approach:  columns table (metadata) + rows table (JSON data blob)
    52→```

    56→## 2. Prisma Schema — Complete
    57→
    58→```prisma
    59→generator client {
    60→  provider = "prisma-client-js"
    61→}
    62→
    63→datasource db {
    64→  provider = "sqlite"
    65→  url      = env("DATABASE_URL")

    68→// ═══════════════════════════════════════════════════════════════
    69→// PILLAR 1: DATA — Dynamic Data Sources, Columns, Rows
    70→// ═══════════════════════════════════════════════════════════════
    71→
    72→/// A user-defined table (e.g., "Products", "Colors", "Categories")
    73→model DataSource {
    74→  id          String    @id @default(cuid())
    75→  name        String    @unique          // "Products", "Colors", etc.
    76→  slug        String    @unique          // URL-safe identifier
    77→  icon        String    @default("table") // Lucide icon name
    78→  color       String    @default("#6366f1") // Accent color for UI
    79→  sortOrder   Int       @default(0)      // Display order in sidebar
    80→  createdAt   DateTime  @default(now())
    81→  updatedAt   DateTime  @updatedAt
    82→
    83→  // Relations
    84→  columns     Column[]
    85→  rows        Row[]
    86→  sourceOf    Relation[] @relation("SourceTable")   // This table is the source in a relation
    87→  targetOf    Relation[] @relation("TargetTable")   // This table is the target in a relation
    88→
    89→  @@map("data_sources")
    90→}
    91→
    92→/// A column definition within a DataSource — pure metadata
    93→model Column {
    94→  id           String     @id @default(cuid())
    95→  dataSourceId String
    96→  name         String                       // "Price", "Image URL", etc.
    97→  slug         String                       // URL-safe: "price", "image_url"
    98→  type         ColumnType  @default(TEXT)
    99→  sortOrder    Int        @default(0)       // Display order in table
   100→  required     Boolean    @default(false)   // Is this column required?
   101→  visible      Boolean    @default(true)    // Show in table view?
   102→  defaultValue String?                      // JSON-encoded default value
   103→
   104→  // For ARRAY type: which child columns compose this array
   105→  arrayChildIds String?                     // JSON array of column IDs: ["col_1","col_2","col_3"]
   106→
   107→  // For RELATION type: which table/column this links to
   108→  relationId   String?                      // FK to Relation record (if type = RELATION)
   109→
   110→  createdAt    DateTime   @default(now())
   111→  updatedAt    DateTime   @updatedAt
   112→
   113→  dataSource   DataSource @relation(fields: [dataSourceId], references: [id], onDelete: Cascade)
   114→
   115→  @@unique([dataSourceId, slug])
   116→  @@map("columns")
   117→}
   118→
   119→/// Column types — determines how data is stored, validated, and rendered
   120→enum ColumnType {
   121→  TEXT          // Short text (single line)
   122→  LONG_TEXT     // Multi-line text / description
   123→  NUMBER        // Numeric (int or float)
   124→  CURRENCY      // Number formatted as price
   125→  IMAGE         // Single image URL
   126→  IMAGE_ARRAY   // JSON array of image URLs (standalone, not composed)
   127→  COLOR         // Color swatch: JSON { name, hex }
   128→  BOOLEAN       // true/false
   129→  DATE          // ISO date string
   130→  URL           // Web URL
   131→  PHONE         // Phone number
   132→  EMAIL         // Email address
   133→  SELECT        // Single select from predefined options
   134→  MULTI_SELECT  // Multi select from predefined options
   135→  ARRAY         // Virtual column composed of other columns (arrayChildIds)
   136→  RELATION      // Links to a row in another DataSource
   137→}
   138→
   139→/// A row of data — stores all cell values in a single JSON object keyed by column ID
   140→model Row {
   141→  id           String     @id @default(cuid())
   142→  dataSourceId String
   143→  data         String     // JSON: { "col_abc123": "value1", "col_def456": 42, ... }
   144→  sortOrder    Int        @default(0)
   145→  createdAt    DateTime   @default(now())
   146→  updatedAt    DateTime   @updatedAt
   147→
   148→  dataSource   DataSource @relation(fields: [dataSourceId], references: [id], onDelete: Cascade)
   149→
   150→  @@map("rows")
   151→}
   152→
   153→/// A relation between two DataSources (like a foreign key)
   154→model Relation {
   155→  id             String     @id @default(cuid())
   156→  name           String                       // "Product → Category"
   157→  sourceColumnId String                       // The column in source table that holds the reference
   158→  sourceTableId  String                       // The source DataSource
   159→  targetTableId  String                       // The target DataSource
   160→  type           RelationType @default(MANY_TO_ONE)
   161→
   162→  sourceColumn   Column     @relation(fields: [sourceColumnId], references: [id], onDelete: Cascade)
   163→  sourceTable    DataSource @relation("SourceTable", fields: [sourceTableId], references: [id], onDelete: Cascade)
   164→  targetTable    DataSource @relation("TargetTable", fields: [targetTableId], references: [id], onDelete: Cascade)
   165→
   166→  @@map("relations")
   167→}
   168→
   169→enum RelationType {
   170→  MANY_TO_ONE    // Many rows in source → one row in target (e.g., products → category)
   171→  ONE_TO_MANY    // One row in source → many rows in target (e.g., category → products)
   172→  MANY_TO_MANY   // Junction (future: via junction table)
   173→}
   174→
   175→/// Select options for SELECT/MULTI_SELECT column types
   176→model SelectOption {
   177→  id         String   @id @default(cuid())
   178→  columnId   String
   179→  label      String
   180→  value      String   // Stored value (can differ from label)
   181→  color      String?  // Hex color for the tag
   182→  sortOrder  Int      @default(0)
   183→
   184→  column     Column   @relation(fields: [columnId], references: [id], onDelete: Cascade)
   185→
   186→  @@unique([columnId, value])
   187→  @@map("select_options")
   188→}

   194→/// A catalog — the top-level entity that ties data + layout + settings
   195→model Catalog {
   196→  id          String    @id @default(cuid())
   197→  name        String    @default("Mon Catalogue")
   198→  slug        String    @unique @default(cuid())
   199→  published   Boolean   @default(false)
   200→  createdAt   DateTime  @default(now())
   201→  updatedAt   DateTime  @updatedAt
   202→
   203→  // Relations
   204→  pages       Page[]
   205→  settings    CatalogSettings?
   206→  theme       Theme?
   207→
   208→  @@map("catalogs")
   209→}
   210→
   211→/// A page within a catalog (e.g., "Home", "All Products", "About")
   212→model Page {
   213→  id          String    @id @default(cuid())
   214→  catalogId   String
   215→  name        String                       // "Home", "Products", "About"
   216→  slug        String                       // "home", "products", "about"
   217→  route       String                       // "/" , "/products", "/about"
   218→  sortOrder   Int       @default(0)
   219→  isHome      Boolean   @default(false)    // Landing page?
   220→  createdAt   DateTime  @default(now())
   221→  updatedAt   DateTime  @updatedAt
   222→
   223→  catalog     Catalog   @relation(fields: [catalogId], references: [id], onDelete: Cascade)
   224→  sections    Section[]
   225→
   226→  @@unique([catalogId, slug])
   227→  @@map("pages")
   228→}
   229→
   230→/// A section on a page — the fundamental layout unit
   231→model Section {
   232→  id         String      @id @default(cuid())
   233→  pageId     String
   234→  type       SectionType
   235→  config     String      @default("{}") // JSON: section-level config (padding, background, etc.)
   236→  sortOrder  Int         @default(0)
   237→  visible    Boolean     @default(true)
   238→  createdAt  DateTime    @default(now())
   239→  updatedAt  DateTime    @updatedAt
   240→
   241→  page       Page        @relation(fields: [pageId], references: [id], onDelete: Cascade)
   242→  components Component[]
   243→
   244→  @@map("sections")
   245→}
   246→
   247→enum SectionType {
   248→  HERO            // Hero banner with image + text
   249→  COLLECTION      // Grid/list of items from a DataSource
   250→  DETAIL_VIEW     // Full product detail (carousel, variants, etc.)
   251→  RICH_TEXT       // Free text block
   252→  IMAGE_BANNER    // Full-width image
   253→  SEPARATOR       // Visual divider
   254→  CAROUSEL_SECTION // Horizontal scroll of items
   255→}
   256→
   257→/// A component within a section — maps data columns to UI elements
   258→model Component {
   259→  id         String    @id @default(cuid())
   260→  sectionId  String
   261→  type       ComponentType
   262→  mappings   String    @default("{}") // JSON: { title: "col_abc", description: "col_def", ... }
   263→  style      String    @default("{}") // JSON: { fontSize: "lg", borderRadius: "md", ... }
   264→  sortOrder  Int       @default(0)
   265→  createdAt  DateTime  @default(now())
   266→  updatedAt  DateTime  @updatedAt
   267→
   268→  section    Section   @relation(fields: [sectionId], references: [id], onDelete: Cascade)
   269→
   270→  @@map("components")
   271→}
   272→
   273→enum ComponentType {
   274→  // Level 1 — Collection level
   275→  CARD_GRID         // Grid of product cards
   276→  CARD_LIST         // List of product cards (horizontal)
   277→  TABLE_VIEW        // Tabular display
   278→
   279→  // Level 2 — Card presentation
   280→  CARD_COVER        // Card with cover image (used within CARD_GRID)
   281→  CARD_INLINE       // Card with inline image
   282→
   283→  // Level 3 — Detail view
   284→  IMAGE_CAROUSEL    // Carousel of images (from array column)
   285→  COLOR_SWATCHES    // Grid of color swatches
   286→  VARIANT_SELECTOR  // Size/variant picker
   287→  FIELD_DISPLAY     // Single field display (label + value)
   288→  ACTION_BUTTON     // CTA button (WhatsApp, etc.)
   289→  RELATED_ITEMS     // Related items from a relation
   290→
   291→  // Misc
   292→  SEARCH_BAR        // Search input
   293→  FILTER_BAR        // Category/filter chips
   294→  BREADCRUMB        // Navigation breadcrumb
   295→}

   387→
   388→### Schema Design Rationale
   389→
   390→| Decision | Why |
   391→|----------|-----|
   392→| **Row.data is JSON string** | SQLite doesn't support JSON columns natively in Prisma, but `String` works fine with `JSON.parse/stringify`. Columns are metadata; rows are a flat JSON blob keyed by `columnId`. |
   393→| **Column.arrayChildIds is JSON** | An ARRAY column groups multiple real columns. The `arrayChildIds` field stores `["col_a","col_b","col_c"]` so we know which sub-columns to compose. |
   394→| **Relation is a first-class entity** | Instead of burying FK info inside column config, Relations are their own table. This makes validation, resolution, and future many-to-many easier. |
   395→| **Component.mappings is JSON** | Each component type has different mapping needs. `CARD_GRID` maps `{ title, description, coverImage }`, while `IMAGE_CAROUSEL` maps `{ images }`. Storing as flexible JSON avoids a table per component type. |
   396→| **Catalog → Page → Section → Component** | Four-level nesting mirrors how visual editors work. This is the same pattern Glide uses: App → Screen → Container → Element. |
   397→| **CatalogSettings is 1:1** | Separate model avoids bloating the Catalog model. One-to-one via `@unique` on `catalogId`. |
   398→| **Theme is separate** | Theme values change independently and may be swapped/shared in future. |

   469→### Key API Endpoints Detail
   470→
   471→#### CSV Import (`POST /api/data-sources/[id]/import`)
   472→
   473→```
   474→Input:  FormData with CSV file
   475→Process:
   476→  1. Parse CSV headers → create Column records (auto-detect type: TEXT/NUMBER/IMAGE)
   477→  2. Parse CSV rows → create Row records with data JSON keyed by column IDs
   478→  3. Return: { imported: N, columns: [...], errors: [...] }
   479→```
   480→
   481→#### Public Catalog Data (`GET /api/public/[catalogSlug]`)
   482→
   483→```
   484→Returns:
   485→{
   486→  catalog: { name, slug, theme, settings },
   487→  pages: [
   488→    {
   489→      name, slug, route,
   490→      sections: [
   491→        {
   492→          type, config,
   493→          components: [
   494→            { type, mappings, style }
   495→          ]
   496→        }
   497→      ]
   498→    }
   499→  ],
   500→  dataSources: {  // only the ones referenced by components
   501→    [slug]: {
   502→      columns: [...],
   503→      rows: [...]  // first page
   504→    }
   505→  }
   506→}
   507→```

   513→### 4.1 Admin Builder — Three-Pillar Layout
   514→
   515→```
   516→<BuilderLayout>
   517→├── <BuilderSidebar>              // Left: pillar navigation
   518→│   ├── <PillarNav>
   519→│   │   ├── <NavItem icon="database" label="Data" />
   520→│   │   ├── <NavItem icon="layout"   label="Layout" />
   521→│   │   └── <NavItem icon="settings" label="Settings" />
   522→│   └── <DataSourceList />        // Quick access to tables (below nav)
   523→│
   524→├── <BuilderMain>                 // Center: active pillar content
   525→│   │
   526→│   ├── DATA PILLAR:
   527→│   │   ├── <DataSourceManager>   // List/manage all data sources
   528→│   │   │   ├── <DataSourceCard>  // Summary card per table
   529→│   │   │   └── <ImportCSVModal>  // CSV import wizard
   530→│   │   │
   531→│   │   └── <DataSourceEditor>    // Single table editor
   532→│   │       ├── <ColumnHeader>    // Column headers with type badges
   533→│   │       │   ├── <ColumnNameCell>
   534→│   │       │   ├── <ColumnTypeCell>
   535→│   │       │   └── <ColumnActionsMenu>  // Edit, delete, add relation/array

   840→### 6.1 CSV Import Flow
   841→
   842→```
   843→User drops CSV file
   844→        │
   845→        ▼
   846→[Frontend: ImportCSVModal]
   847→        │  POST /api/data-sources/[id]/import
   848→        │  FormData { file }
   849→        ▼
   850→[API: import/route.ts]
   851→        │
   852→        ├── 1. Read CSV with papaparse
   853→        │      Parse headers → detect types
   854→        │        - If header contains "image"/"photo"/"img" → IMAGE
   855→        │        - If header contains "price"/"prix"/"cost"  → CURRENCY
   856→        │        - If all values are numeric                  → NUMBER
   857→        │        - Otherwise                                  → TEXT
   858→        │
   859→        ├── 2. Create DataSource (if new) or use existing
   860→        │
   861→        ├── 3. Create Column records
   862→        │      for each CSV header:
   863→        │        db.column.create({ name, slug, type, dataSourceId })
   864→        │
   865→        ├── 4. Create Row records
   866→        │      for each CSV row:
   867→        │        const data = {};
   868→        │        for each header/column:
   869→        │          data[column.id] = csvRow[header];
   870→        │        db.row.create({ dataSourceId, data: JSON.stringify(data) })
   871→        │
   872→        └── 5. Return { imported: N, columns: [...], skipped: M }
   873→                │
   874→                ▼
   875→[Frontend: Update data-store]
   876→        │  setColumns(newColumns)
   877→        │  setRows(newRows)
   878→        ▼
   879→[TableGrid re-renders]
   880→```

  1047→### 6.6 Relation Resolution Flow
  1048→
  1049→```
  1050→DataSource "Products" has column "categorie" of type RELATION
  1051→        │  relationId → Relation {
  1052→        │    sourceColumnId: "col_categorie",
  1053→        │    sourceTableId: "ds_products",
  1054→        │    targetTableId: "ds_categories",
  1055→        │    type: MANY_TO_ONE
  1056→        │  }
  1057→        │
  1058→        ▼
  1059→When rendering a product row:
  1060→        │
  1061→        row.data["col_categorie"] = "row_xyz789"  // stores target row ID
  1062→        │
  1063→        ▼
  1064→[resolveRelation(row, relationColumn)]
  1065→        │
  1066→        const targetRowId = row.data[relationColumn.id];
  1067→        const targetRow = await db.row.findUnique({ where: { id: targetRowId } });
  1068→        const targetColumns = await db.column.findMany({
  1069→          where: { dataSourceId: relation.targetTableId }
  1070→        });
  1071→        const resolved = {};
  1072→        for (const col of targetColumns) {
  1073→          resolved[col.slug] = JSON.parse(targetRow.data)[col.id];
  1074→        }
  1075→        // resolved = { name: "Abayas", slug: "abayas", image: "..." }
  1076→        │
  1077→        ▼
  1078→[Frontend receives _relations: { col_categorie: { name: "Abayas", ... } }]
  1079→```

  1153→function detectColumnType(header: string, values: string[]): ColumnType {
  1154→  const h = header.toLowerCase();
  1155→  if (/image|img|photo|picture|pic/.test(h)) return 'IMAGE';
  1156→  if (/price|prix|cost|montant|amount/.test(h)) return 'CURRENCY';
  1157→  if (/color|couleur|colour/.test(h)) return 'COLOR';
  1158→  if (/stock|quantity|quantité/.test(h)) return 'NUMBER';
  1159→  if (/available|disponible|active/.test(h)) return 'BOOLEAN';
  1160→  if (/description|desc|detail/.test(h)) return 'LONG_TEXT';
  1161→
  1162→  // Heuristic: if 80%+ values are numeric
  1163→  const numericCount = values.filter(v => !isNaN(Number(v)) && v.trim() !== '').length;
  1164→  if (numericCount / values.length > 0.8) return 'NUMBER';
  1165→
  1166→  return 'TEXT';
  1167→}

  1189→### 7.5 Column-to-Component Mapping Resolution
  1190→
  1191→This is the core rendering engine. Given a component's `mappings` JSON and a row's `data` JSON, resolve what to display:
  1192→
  1193→```typescript
  1194→// src/lib/resolve-mappings.ts
  1195→interface MappingConfig {
  1196→  dataSourceId: string;
  1197→  title?: string;        // column ID
  1198→  subtitle?: string;     // column ID
  1199→  description?: string;  // column ID
  1200→  coverImage?: string;   // column ID
  1201→  price?: string;        // column ID
  1202→  carouselImages?: string;  // column ID (must be ARRAY or IMAGE_ARRAY type)
  1203→  colorVariants?: string;   // column ID
  1204→  sizeVariants?: string;    // column ID
  1205→  filterBy?: string;        // column ID (for category filter)
  1206→  sortBy?: string;          // column ID
  1207→}
  1208→
  1209→function resolveCardData(mapping: MappingConfig, row: Row, columns: Column[]) {
  1210→  const data = JSON.parse(row.data);
  1211→  const colMap = Object.fromEntries(columns.map(c => [c.id, c]));
  1212→
  1213→  return {
  1214→    id: row.id,
  1215→    title: data[mapping.title ?? ''] ?? '',
  1216→    subtitle: resolveValue(data, mapping.subtitle, colMap),
  1217→    description: resolveValue(data, mapping.description, colMap),
  1218→    coverImage: resolveValue(data, mapping.coverImage, colMap),
  1219→    price: mapping.price ? {
  1220→      value: Number(data[mapping.price] ?? 0),
  1221→      currency: colMap[mapping.price]?.type === 'CURRENCY',
  1222→    } : null,
  1223→  };
  1224→}
  1225→
  1226→function resolveDetailData(mapping: MappingConfig, row: Row, columns: Column[]) {
  1227→  const data = JSON.parse(row.data);
  1228→  const colMap = Object.fromEntries(columns.map(c => [c.id, c]));
  1229→  const card = resolveCardData(mapping, row, columns);
  1230→
  1231→  // Resolve array columns
  1232→  const carouselImages = mapping.carouselImages
  1233→    ? resolveArrayColumn(data, mapping.carouselImages, colMap)
  1234→    : [];
  1235→
  1236→  const colorVariants = mapping.colorVariants
  1237→    ? JSON.parse(data[mapping.colorVariants] || '[]')
  1238→    : [];
  1239→
  1240→  const sizeVariants = mapping.sizeVariants
  1241→    ? JSON.parse(data[mapping.sizeVariants] || '[]')
  1242→    : [];
  1243→
  1244→  return { ...card, carouselImages, colorVariants, sizeVariants };
  1245→}
  1246→
  1247→function resolveArrayColumn(
  1248→  data: Record<string, unknown>,
  1249→  arrayColumnId: string,
  1250→  colMap: Record<string, Column>
  1251→): string[] {
  1252→  const col = colMap[arrayColumnId];
  1253→  if (!col) return [];
  1254→
  1255→  if (col.type === 'IMAGE_ARRAY') {
  1256→    return JSON.parse((data[arrayColumnId] as string) || '[]');
  1257→  }
  1258→
  1259→  if (col.type === 'ARRAY') {
  1260→    const childIds: string[] = JSON.parse(col.arrayChildIds || '[]');
  1261→    return childIds.map(id => data[id]).filter(Boolean) as string[];
  1262→  }
  1263→
  1264→  return [];
  1265→}
  1266→```

  1303→### 7.7 Debounced Batch Save for Inline Editing
  1304→
  1305→```typescript
  1306→// src/lib/stores/data-store.ts (excerpt)
  1307→let flushTimeout: NodeJS.Timeout | null = null;
  1308→
  1309→markDirty: (rowId, colId, value) => {
  1310→  set(state => {
  1311→    const newDirty = new Map(state.dirtyCells);
  1312→    newDirty.set(`${rowId}:${colId}`, value);
  1313→
  1314→    // Optimistic local update
  1315→    const newRows = state.rows.map(r => {
  1316→      if (r.id !== rowId) return r;
  1317→      const data = JSON.parse(r.data);
  1318→      data[colId] = value;
  1319→      return { ...r, data: JSON.stringify(data) };
  1320→    });
  1321→
  1322→    return { dirtyCells: newDirty, rows: newRows };
  1323→  });
  1324→
  1325→  // Debounce flush
  1326→  if (flushTimeout) clearTimeout(flushTimeout);
  1327→  flushTimeout = setTimeout(() => get().flushDirty(), 500);
  1328→},
  1329→
  1330→flushDirty: async () => {
  1331→  const { dirtyCells, rows } = get();
  1332→  if (dirtyCells.size === 0) return;
  1333→
  1334→  // Group by rowId
  1335→  const byRow = new Map<string, Record<string, string>>();
  1336→  for (const [key, value] of dirtyCells) {
  1337→    const [rowId, colId] = key.split(':');
  1338→    if (!byRow.has(rowId)) byRow.set(rowId, {});
  1339→    byRow.get(rowId)![colId] = value;
  1340→  }
  1341→
  1342→  // Batch PATCH each row
  1343→  await Promise.all(
  1344→    Array.from(byRow.entries()).map(([rowId, updates]) =>
  1345→      fetch(`/api/data-sources/${get().activeDataSourceId}/rows/${rowId}`, {
  1346→        method: 'PATCH',
  1347→        headers: { 'Content-Type': 'application/json' },
  1348→        body: JSON.stringify({ data: updates }),
  1349→      })
  1350→    )
  1351→  );
  1352→
  1353→  set({ dirtyCells: new Map() });
  1354→},
  1355→```

  1357→### 7.8 Relation Resolution on the API Side
  1358→
  1359→```typescript
  1360→// src/app/api/public/[catalogSlug]/product/[rowId]/route.ts
  1361→export async function GET(req: NextRequest, { params }: { params: { rowId: string } }) {
  1362→  const row = await db.row.findUnique({ where: { id: params.rowId } });
  1363→  if (!row) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  1364→
  1365→  const data = JSON.parse(row.data);
  1366→
  1367→  // Resolve all RELATION columns
  1368→  const columns = await db.column.findMany({
  1369→    where: { dataSourceId: row.dataSourceId, type: 'RELATION' },
  1370→    include: { relation: true },
  1371→  });
  1372→
  1373→  const resolvedRelations: Record<string, any> = {};
  1374→
  1375→  for (const col of columns) {
  1376→    if (!col.relation) continue;
  1377→    const targetRowId = data[col.id];
  1378→    if (!targetRowId) continue;
  1379→
  1380→    const targetRow = await db.row.findUnique({ where: { id: targetRowId } });
  1381→    if (!targetRow) continue;
  1382→
  1383→    const targetColumns = await db.column.findMany({
  1384→      where: { dataSourceId: col.relation.targetTableId },
  1385→    });
  1386→
  1387→    const targetData = JSON.parse(targetRow.data);
  1388→    const resolved: Record<string, any> = {};
  1389→    for (const tc of targetColumns) {
  1390→      resolved[tc.slug] = targetData[tc.id];
  1391→    }
  1392→    resolvedRelations[col.id] = resolved;
  1393→  }
  1394→
  1395→  return NextResponse.json({
  1396→    ...data,
  1397→    _id: row.id,
  1398→    _relations: resolvedRelations,
  1399→  });
  1400→}
